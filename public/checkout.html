function generateSlots(){ const slotSelect = document.getElementById('slotSelect'); slotSelect.innerHTML = ''; // placeholder prompt const pc = document.getElementById('postcode').value.trim(); const userZone = getZoneByPostcode(pc); if(!userZone) return; const now = new Date(); const earliest = new Date(now.getTime()+8*3600*1000); // 8-hour prep let day = new Date(); let slotsAdded = 0; // Add "Anytime" as first selectable slot const anytimeOption = document.createElement('option'); anytimeOption.value = 'anytime'; anytimeOption.textContent = 'Anytime'; slotSelect.appendChild(anytimeOption); while(slotsAdded < 6){ // generate six slots if(day.getDay()!==0){ // skip Sundays const rotation = DAYS_ROTATION[(day.getDay()-1)%DAYS_ROTATION.length]; // rotate days for(let i=0;i<3;i++){ if(rotation[i]===userZone){ const time = TIME_BLOCKS[i]; const slotDate = new Date(day); const slotStartHour = parseInt(time.split(':')[0]); slotDate.setHours(slotStartHour,0,0,0); if(slotDate>=earliest && slotsAdded<6){ const option = document.createElement('option'); option.value = `${day.toISOString().slice(0,10)}|${time}`; option.textContent = `${day.toDateString()} — ${time}`; slotSelect.appendChild(option); slotsAdded++; } } } } day.setDate(day.getDate()+1); } }
